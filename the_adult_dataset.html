# Calculating the skewness of male and female population
def skewness(x):
    res = 0

    m = x.mean()
    s = x.std()

    for i in x:
        res += (i-m) ** 3
    res /= (len(x) *  (s ** 3))
    return res
print(f"skewness of male population is = {skewness(ml2_age)}")
print(f"skewness of female population is = {skewness(fm2_age)}")
